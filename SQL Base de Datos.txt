-- =========================

-- =========================

CREATE TABLE [dbo].[CertificadoSalario](
    [idCertificadoSalario] [int] IDENTITY(1,1) NOT NULL,
    [fechaSolicitud] [date] NULL,
    [cedulaEmpleado] [char](20) NULL,
    [documentoGenerado] [bit] NOT NULL,
    CONSTRAINT PK_CertificadoSalario PRIMARY KEY CLUSTERED ([idCertificadoSalario] ASC)
)
GO

CREATE TABLE [dbo].[FCL](
    [idFCL] [int] IDENTITY(1,1) NOT NULL,
    [fechaSolicitud] [date] NULL,
    [cedulaEmpleado] [char](20) NULL,
    [documentoGenerado] [bit] NOT NULL,
    CONSTRAINT PK_FCL PRIMARY KEY CLUSTERED ([idFCL] ASC)
)
GO

CREATE TABLE [dbo].[GrupoObjetivo](
    [idGrupoObjetivo] [int] IDENTITY(1,1) NOT NULL,
    [nombreGrupoObjetivo] [varchar](15) NULL,
    CONSTRAINT PK_GrupoObjetivo PRIMARY KEY CLUSTERED ([idGrupoObjetivo] ASC)
)
GO

CREATE TABLE [dbo].[PoliticaAniosExperiencia](
    [idPoliticaAniosExperiencia] [int] IDENTITY(1,1) NOT NULL,
    [aniosDesde] [int] NULL,
    [aniosHasta] [int] NULL,
    [diasPermitidos] [int] NULL,
    [condicionVacaciones] [varchar](10) NULL,
    [fechaInicioVigencia] [date] NULL,
    [fechaFinVigencia] [date] NULL,
    [activo] [bit] NULL,
    [idGrupoObjetivo] [int] NULL,
    [idRolAprobacion] [int] NULL,
    CONSTRAINT PK_PoliticaAniosExperiencia PRIMARY KEY CLUSTERED ([idPoliticaAniosExperiencia] ASC)
)
GO

CREATE TABLE [dbo].[PoliticaCumpleanios](
    [idPoliticaCumpleanios] [int] IDENTITY(1,1) NOT NULL,
    [rangoDiasDisponibles] [int] NULL,
    [soloMesCumpleanios] [bit] NULL,
    [fechaInicioVigencia] [date] NULL,
    [fechaFinVigencia] [date] NULL,
    [activo] [bit] NULL,
    [idGrupoObjetivo] [int] NULL,
    CONSTRAINT PK_PoliticaCumpleanios PRIMARY KEY CLUSTERED ([idPoliticaCumpleanios] ASC)
)
GO

CREATE TABLE [dbo].[Rol](
    [idRol] [int] IDENTITY(1,1) NOT NULL,
    [nombreRol] [varchar](50) NULL,
    CONSTRAINT PK_Rol PRIMARY KEY CLUSTERED ([idRol] ASC)
)
GO

CREATE TABLE [dbo].[SolicitudPermiso](
    [idSolicitudPermiso] [int] IDENTITY(1,1) NOT NULL,
    [fechaSolicitud] [date] NULL,
    [fechaAusencia] [date] NULL,
    [horaEntrada] [time](7) NULL,
    [horaSalida] [time](7) NULL,
    [estado] [varchar](20) NULL,
    [numeroSemana] [int] NULL,
    [cedulaEmpleado] [char](20) NULL,
    [idTipoPermiso] [int] NULL,
    [cedulaAprobador] [char](20) NULL,
    CONSTRAINT PK_SolicitudPermiso PRIMARY KEY CLUSTERED ([idSolicitudPermiso] ASC)
)
GO

CREATE TABLE [dbo].[SolicitudVacaciones](
    [idSolicitudVacaciones] [int] IDENTITY(1,1) NOT NULL,
    [fechaSolicitud] [date] NULL,
    [fechaInicio] [date] NULL,
    [fechaFin] [date] NULL,
    [totalDias] [int] NULL,
    [estado] [varchar](20) NULL,
    [numeroSemana] [int] NULL,
    [cedulaEmpleado] [char](20) NULL,
    [cedulaAprobador] [char](20) NULL,
    CONSTRAINT PK_SolicitudVacaciones PRIMARY KEY CLUSTERED ([idSolicitudVacaciones] ASC)
)
GO

CREATE TABLE [dbo].[TipoPermiso](
    [idTipoPermiso] [int] IDENTITY(1,1) NOT NULL,
    [nombreTipoPermiso] [varchar](40) NULL,
    [activo] [bit] NULL,
    CONSTRAINT PK_TipoPermiso PRIMARY KEY CLUSTERED ([idTipoPermiso] ASC)
)
GO

CREATE TABLE [dbo].[Usuario](
    [idUsuario] [int] IDENTITY(1,1) NOT NULL,
    [correoLogin] [varchar](100) NOT NULL,
    [contrasenaHash] [varchar](255) NOT NULL,
    [cedulaEmpleado] [char](20) NULL,
    [idRol] [int] NULL,
    CONSTRAINT PK_Usuario PRIMARY KEY CLUSTERED ([idUsuario] ASC),
    CONSTRAINT UQ_Usuario UNIQUE NONCLUSTERED ([correoLogin] ASC)
)
GO
